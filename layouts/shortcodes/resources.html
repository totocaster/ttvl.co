{{- /* Resources manifest: files and external destinations in one row grammar. Inner syntax is a markdown-style list; trailing text after a link becomes a note: - [Title](/downloads/file.pdf) optional note - [Title](https://example.com/...) Renders as rows (partials/row.html): the type (PDF, STL) or the destination (GitHub, Thangs, ...) as the meta column, the title with the note as its dek, and the tail: a file's size in Mono with ↓, or ↗ for a page elsewhere. File rows read type and size from static/ at build time. docs/styleguide.md, Page by Page: project page. */ -}} {{- $destinations := dict "github.com" "GitHub" "community.obsidian.md" "Obsidian Community" "thangs.com" "Thangs" "printables.com" "Printables" -}}
    {{- range split (trim .Inner "\n\r\t ") "\n" }} {{- $m := findRE `^\s*[-*]\s*\[[^\]]+\]\([^)]+\).*$` . 1 }} {{- if $m }} {{- $line := index $m 0 }} {{- $title := replaceRE `^\s*[-*]\s*\[([^\]]+)\].*$` "$1" $line }} {{- $url := replaceRE `^\s*[-*]\s*\[[^\]]+\]\(([^)]+)\).*$` "$1" $line }} {{- $note := replaceRE `^\s*[-*]\s*\[[^\]]+\]\([^)]+\)\s*[-–:,]*\s*(.*)$` "$1" $line }} {{- $dek := "" }} {{- with $note }}{{ $dek = . | markdownify }}{{ end }} {{- if hasPrefix $url "/" }} {{- $ext := strings.TrimPrefix "." (path.Ext $url) | upper }} {{- $tail := "↓" }} {{- $fsPath := printf "static%s" $url }} {{- if os.FileExists $fsPath }} {{- $bytes := (os.Stat $fsPath).Size }} {{- $size := "" }} {{- if ge $bytes 1048576 }}{{ $size = printf `%.1f MB` (div (float $bytes) 1048576) }}{{ else }}{{ $size = printf `%.0f KB` (div (float $bytes) 1024) }}{{ end }} {{- $tail = printf `%s ↓` $size }} {{- end }} {{ partial "row.html" (dict "meta" $ext "url" $url "title" $title "dek" $dek "tail" ($tail | safeHTML) "download" true "class" "resource-row--file") }} {{- else }} {{- $host := strings.TrimPrefix "www." (urls.Parse $url).Host }} {{ partial "row.html" (dict "meta" (index $destinations $host | default $host) "url" $url "title" $title "dek" $dek "tail" "↗" "class" "resource-row--external") }} {{- end }} {{- end }} {{- end }}
