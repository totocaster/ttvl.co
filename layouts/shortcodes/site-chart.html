{{- /* A figure Hugo draws from the site's own pages at every build, so it never goes stale, like the research pulse. docs/figures.md, Making Figures. This one counts the pages published each year, the Log's months apart from everything else. {{< site-chart caption="…" >}} Drawn in the drawing palette and wrapped like a drawing, so it follows the theme (components/_drawing.scss); feeds get its summary as a line of text (data-live). */ -}} {{- $pages := where site.RegularPages "Type" "ne" "page" -}} {{- $first := 2018 -}} {{- $last := now.Year -}} {{- $years := seq $first $last -}} {{- $rows := slice -}} {{- $max := 1 -}} {{- range $years -}} {{- $y := . -}} {{- $log := 0 }}{{ $other := 0 -}} {{- range $pages -}} {{- if eq .Date.Year $y -}} {{- if eq .Section "log" }}{{ $log = add $log 1 }}{{ else }}{{ $other = add $other 1 }}{{ end -}} {{- end -}} {{- end -}} {{- if gt (add $log $other) $max }}{{ $max = add $log $other }}{{ end -}} {{- $rows = $rows | append (dict "year" $y "log" $log "other" $other) -}} {{- end -}} {{- $w := 760.0 }}{{ $top := 28.0 }}{{ $base := 176.0 -}} {{- $unit := div (sub $base $top) (float $max) -}} {{- $slot := div $w (float (len $rows)) -}} {{- $col := 24.0 -}} {{- $label := slice -}} {{- range $rows }}{{ $label = $label | append (printf "%d: %d" .year (add .log .other)) }}{{ end -}}
