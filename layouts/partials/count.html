{{- /* A count as words after the number: "13 pages", "1 page", with the number in Mono. Pass a dict with "n", "one", and "many"; "live" wraps it as .count[data-singular][data-plural] so category-filter.js can rewrite it. A page carries one count at most, in its head rail. Returns HTML. docs/styleguide.md, Block. */ -}} {{- $html := printf `%d %s` (int .n) (cond (eq (int .n) 1) .one .many) -}} {{- if .live }}{{ $html = printf `%s` .one .many $html }}{{ end -}} {{- return ($html | safeHTML) -}}
