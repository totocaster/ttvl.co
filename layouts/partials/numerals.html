{{- /* Sets the numbers in a short label in Mono: escapes the text, wraps every number that starts a word in , and turns Markdown links [text](url) into links. For rail meta lines written in front matter or shortcode parameters ("Updated 2026-09 · Not awards"). Returns HTML. */ -}} {{- $s := htmlEscape (string .) -}} {{- $s = replaceRE `(^|[\s(])(\d(?:[\d\-–.:,/]*\d)?)` `${1}${2}` $s -}} {{- $s = replaceRE `\[([^\]]+)\]\(([^)\s]+)\)` `${1}` $s -}} {{- return ($s | safeHTML) -}}
