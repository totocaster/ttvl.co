{{- /* Terminal block for shell code fences. Each command line's $ prompt is a quiet copy button for that command (continuation lines after a trailing backslash are included and carry no prompt; # comment lines render faint). The copy script loads once per page, only when a shell block is present. */ -}}
{{- $cont := false }} {{- range split (trim .Inner "\n") "\n" }} {{- $line := . }} {{- $trimmed := trim $line " \t" }} {{- if $cont }} {{ $line }} {{- else if eq $trimmed "" }} {{- else if hasPrefix $trimmed "#" }} {{ $line }} {{- else }} {{ $line }} {{- end }} {{- $cont = hasSuffix (strings.TrimRight " " $line) "\\" }} {{- end }}
{{- if not (.Page.Store.Get "shellCopyJS") }} {{- .Page.Store.Set "shellCopyJS" true }} {{- $js := resources.Get "js/shell-copy.js" | resources.Minify | resources.Fingerprint }} {{- end }}
