{{- /* A page's canonical doors, from `resources:` frontmatter (title + url). File chips carry their size; external chips carry the ↗ mark via CSS. */ -}} {{ with .Params.resources }}

{{- range . }} {{- $url := .url }} {{- if hasPrefix $url "/" }} {{- $size := "" }} {{- $fsPath := printf "static%s" $url }} {{- if os.FileExists $fsPath }} {{- $bytes := (os.Stat $fsPath).Size }} {{- if ge $bytes 1048576 }}{{ $size = printf "%.1f MB" (div (float $bytes) 1048576) }}{{ else }}{{ $size = printf "%.0f KB" (div (float $bytes) 1024) }}{{ end }} {{- end }} {{ .title }}{{ with $size }} {{ . }}{{ end }} {{- else }} {{ .title }} {{- end }} {{- end }}

{{ end }}
