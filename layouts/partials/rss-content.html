{{- /* RSS readers do not load the site's CSS or JavaScript. Convert only the shell renderer's known markup to semantic code blocks, preserving escaped command text and explicit line breaks without interactive copy controls. */ -}} {{- $content := .Content -}} {{- range findRE `(?s)
.*?
` $content -}} {{- $block := replaceRE `^
\s*|
$` "" . -}} {{- $block = replaceRE `` "$" $block -}} {{- $block = replaceRE `\s*` "\n" $block -}} {{- $block = replaceRE `` "" $block -}} {{- $content = replace $content . (printf "
%s
" $block) -}} {{- end -}} {{- $content = replaceRE `]*src="[^"]*/js/shell-copy\.[^"]+\.js"[^>]*>` "" $content -}} {{- return $content -}}
