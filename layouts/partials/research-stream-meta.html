{{- /* A stream's tail in the hub's Streams rows: its entry count and the date of its newest entry, numbers in Mono ("24 entries · 2026-10-06"). Returns HTML. */ -}} {{- $entries := .RegularPages.ByDate.Reverse -}} {{- $parts := slice (partial "count.html" (dict "n" (len $entries) "one" "entry" "many" "entries")) -}} {{- with $entries -}} {{- $d := (index . 0).Date.Format "2006-01-02" -}} {{- $parts = $parts | append (printf `%s` $d $d) -}} {{- end -}} {{- return (delimit $parts " · " | safeHTML) -}}
