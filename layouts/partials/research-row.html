{{- /* One index row for a research entry: date, title, and a meta tail. Takes dict "page" plus booleans "stream", "number", "category" choosing what the tail names. data-category is the stream's folder name, for the hub's stream filter; the stream name carries data-filter-show="all" so the filter hides it once every row shares the same stream. */ -}} {{- $p := .page -}} {{- $rest := slice -}} {{- if .number }}{{ $rest = $rest | append (partialCached "research-number.html" $p $p.Path) }}{{ end -}} {{- if .category }}{{ with partial "research-category.html" $p }}{{ $rest = $rest | append . }}{{ end }}{{ end -}} {{- $tail := "" -}} {{- if or .stream $rest -}} {{- $html := "" -}} {{- if .stream }}{{ $html = printf `%s%s` (htmlEscape $p.CurrentSection.Title) (cond (gt (len $rest) 0) " · " "") }}{{ end -}} {{- $tail = printf "%s%s" $html (delimit $rest " · ") | safeHTML -}} {{- end -}} {{ partial "row.html" (dict "meta" (printf `%s` ($p.Date.Format "2006-01-02") ($p.Date.Format "2006-01-02") | safeHTML) "url" $p.RelPermalink "title" $p.Title "tail" $tail "category" $p.CurrentSection.File.ContentBaseName ) }}
