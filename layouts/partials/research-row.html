{{- /* One row for a research entry: the date as the meta, the title, and a tail naming the entry's number and kind ("024 · Question"). Takes dict "page" plus booleans "stream", "number", "category" choosing what the tail names; with none of them the row has no tail (home lists). docs/styleguide.md, Rows. */ -}} {{- $p := .page -}} {{- $parts := slice -}} {{- if .stream }}{{ $parts = $parts | append (htmlEscape $p.CurrentSection.Title) }}{{ end -}} {{- if .number }}{{ $parts = $parts | append (printf `%s` (partialCached "research-number.html" $p $p.Path)) }}{{ end -}} {{- if .category }}{{ with partial "research-category.html" $p }}{{ $parts = $parts | append (htmlEscape .) }}{{ end }}{{ end -}} {{- $d := $p.Date.Format "2006-01-02" -}} {{ partial "row.html" (dict "meta" (printf `%s` $d $d | safeHTML) "url" $p.RelPermalink "title" $p.Title "tail" (delimit $parts " · " | safeHTML) ) }}
