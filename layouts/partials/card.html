{{- /* A card: a 16:9 tile, one label line, the title, the dek. Pass "page"; pass "variant" "trace" for trace pages, whose description, image, and kind live outside project front matter. The label line: projects read "date · hub" (the hub is the title, or filter_label, of the page named by project.category, so Project Humane reads "Tools"); ongoing work reads "Since YYYY · hub"; traces read "date · kind". Optional project.since (a year) and project.kind (a word) override the year and the hub for work without a category. docs/styleguide.md, Cards. */ -}} {{- $page := .page -}} {{- $variant := .variant | default "project" -}} {{- $img := "" -}} {{- $desc := "" -}} {{- $category := "" -}} {{- $label := slice -}} {{- if eq $variant "trace" -}} {{- $img = $page.Params.image -}} {{- $desc = $page.Params.description -}} {{- $category = $page.Params.trace.kind -}} {{- with $page.Date }}{{ if not .IsZero }}{{ $label = $label | append (printf `%s` (.Format "2006-01-02") (.Format "2006-01-02")) }}{{ end }}{{ end -}} {{- with $page.Params.trace.kind }}{{ $label = $label | append (htmlEscape .) }}{{ end -}} {{- else -}} {{- $proj := $page.Params.project | default (dict) -}} {{- $img = $proj.image -}} {{- $desc = or $proj.description $page.Params.description -}} {{- with $proj.category }}{{ $category = trim . "/" }}{{ end -}} {{- if eq (printf "%v" $proj.year) "ongoing" -}} {{- $since := $proj.since -}} {{- if not $since -}} {{- with $page.RegularPages.ByDate }}{{ $since = (index . 0).Date.Year }}{{ else }}{{ if not $page.Date.IsZero }}{{ $since = $page.Date.Year }}{{ end }}{{ end -}} {{- end -}} {{- $label = $label | append (cond (ne (printf "%v" $since) "") (printf `Since %v` $since) "Ongoing") -}} {{- else if not $page.Date.IsZero -}} {{- $label = $label | append (printf `%s` ($page.Date.Format "2006-01") ($page.Date.Format "2006-01")) -}} {{- end -}} {{- with $proj.kind -}} {{- $label = $label | append (htmlEscape .) -}} {{- else -}} {{- with $proj.category }}{{ with site.GetPage . }}{{ $label = $label | append (htmlEscape (.Params.filter_label | default .Title)) }}{{ end }}{{ end -}} {{- end -}} {{- end }}
