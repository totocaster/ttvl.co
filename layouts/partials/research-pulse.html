{{- /* A stream's activity over the last 13 weeks, one mark per day, oldest at left, counted back from the build. A day with entries stands taller the more it holds (capped at six); an empty day is a low edge-colored stub, so the strip reads as a timeline. The newest marked day takes the signal while it is less than a week old. Sits on the stream's switch on the hub, between its count and its dek; the legend under the switches explains it. */ -}} {{- $days := 91 -}} {{- $counts := dict -}} {{- range .RegularPages -}} {{- $d := .Date.Format "2006-01-02" -}} {{- $counts = merge $counts (dict $d (add (index $counts $d | default 0) 1)) -}} {{- end -}} {{- $levels := slice -}} {{- $total := 0 -}} {{- $newest := -1 -}} {{- range $i := seq 0 (sub $days 1) -}} {{- $d := (now.AddDate 0 0 (sub $i (sub $days 1))).Format "2006-01-02" -}} {{- $n := index $counts $d | default 0 -}} {{- $total = add $total $n -}} {{- if gt $n 0 }}{{ $newest = $i }}{{ end -}} {{- $levels = $levels | append (cond (gt $n 6) 6 $n) -}} {{- end -}} {{- $label := printf "%d %s in the last %d weeks" $total (cond (eq $total 1) "entry" "entries") (div $days 7) -}} {{- with .RegularPages.ByDate.Reverse }}{{ $label = printf "%s, the latest on %s" $label ((index . 0).Date.Format "2006-01-02") }}{{ end -}} {{- range $i, $l := $levels }}{{ end -}}
