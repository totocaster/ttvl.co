{{- /* An article's body: one block with an empty rail, prose in the column. Pass the rendered content (HTML). Blocks that shortcodes emitted inside it stand beside it; body blocks left empty by them are dropped. A page that calls the toc shortcode gets its table of contents in the body block's rail (sticky on wide screens): the section shortcodes it is built from (About), else its Markdown headings. The prose then carries no heading numerals, since the list numbers the sections. A lamp before the entry whose section is in view follows the reader (toc-spy.js). docs/styleguide.md, Block and Prose. */ -}} {{- $seams := partialCached "block-seams.html" "seams" -}} {{- $p := page -}} {{- $open := $seams.open -}} {{- $toc := "" -}} {{- if $p.Store.Get "toc" -}} {{- with $p.Store.Get "sections" -}} {{- $items := "" -}} {{- range . }}{{ $items = printf `%s
  • %s
    • ` $items .id (htmlEscape .title) }}{{ end -}} {{- $toc = printf `
      %s
    ` $items -}} {{- else -}} {{- $toc = string $p.TableOfContents -}} {{- end -}} {{- end -}} {{- /* The body itself as the first entry, when the toc shortcode names it. */ -}} {{- $bodyID := "" -}} {{- with $p.Store.Get "tocIntro" -}} {{- $bodyID = anchorize . -}} {{- $toc = replaceRE `
      ` (printf `
      • %s
        • ` $bodyID (htmlEscape .)) $toc 1 -}} {{- end -}} {{- with $toc -}} {{- $open = printf `

        %s

        %s
        ` (cond (ne $bodyID "") (printf ` id="%s"` $bodyID) "") (htmlEscape ($p.Store.Get "tocTitle" | default "Contents")) . -}} {{- end -}} {{- $html := printf "%s%s%s" $open (string .) $seams.close -}} {{- /* The lamp that follows the reader down the contents (toc-spy.js). */ -}} {{- if $toc -}} {{- $spy := resources.Get "js/toc-spy.js" | resources.Minify | resources.Fingerprint -}} {{- $html = printf `%s` $html $spy.RelPermalink $spy.Data.Integrity -}} {{- end -}} {{- $html = replaceRE (printf `%s\s*%s` $seams.open $seams.close) "" $html -}} {{- return ($html | safeHTML) -}}
